        // --- SYNTHESIZED BELL CHIME TONE ---
        function playBellChime() {
            initAudioContext();
            const now = audioCtx.currentTime;

            // Dual tone frequencies (E5 and G#5 harmonic chime)
            const freqs = [659.25, 830.61];

            freqs.forEach(freq => {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();

                osc.type = 'sine';
                osc.frequency.setValueAtTime(freq, now);

                // Smooth exponential decay (bell envelope)
                gain.gain.setValueAtTime(0.3, now);
                gain.gain.exponentialRampToValueAtTime(0.0001, now + 2.5);

                osc.connect(gain);
                gain.connect(audioCtx.destination);

                osc.start(now);
                osc.stop(now + 2.5);
            });

        function triggerConfetti() {

            animateConfetti();

        function animateConfetti() {
            cCtx.clearRect(0, 0, confettiCanvas.width, confettiCanvas.height);
            let activeCount = 0;

            confettiParticles.forEach(p => {
                p.vx *= p.drag;
                p.vy += p.gravity;
                p.x += p.vx;
                p.y += p.vy;
                p.rotation += p.vRot;
                p.opacity -= 0.006;

                if (p.opacity > 0) {
                    activeCount++;
                    cCtx.save();
                    cCtx.globalAlpha = Math.max(0, p.opacity);
                    cCtx.translate(p.x, p.y);
                    cCtx.rotate(p.rotation);
                    cCtx.fillStyle = p.color;
                    cCtx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size);
                    cCtx.restore();
                }
            });

            if (activeCount > 0) {
                confettiAnimationId = requestAnimationFrame(animateConfetti);
            } else {
                cCtx.clearRect(0, 0, confettiCanvas.width, confettiCanvas.height);
            }
        }

        function playPresetSoundscape(type) {
            initAudioContext();
            if (presetAudioSource) {
                presetAudioSource.stop();
                presetAudioSource = null;
            }
            currentPreset = type;
            if (type === 'off') return;

            const bufferSize = audioCtx.sampleRate * 2;
            const noiseBuffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
            const output = noiseBuffer.getChannelData(0);

            for (let i = 0; i < bufferSize; i++) {
                output[i] = Math.random() * 2 - 1;
            }

            const whiteNoise = audioCtx.createBufferSource();
            whiteNoise.buffer = noiseBuffer;
            whiteNoise.loop = true;

            const filter = audioCtx.createBiquadFilter();
            filter.type = type === 'rain' ? 'lowpass' : 'bandpass';
            filter.frequency.value = type === 'rain' ? 800 : 400;

            whiteNoise.connect(filter);
            filter.connect(audioCtx.destination);
            whiteNoise.start();
            presetAudioSource = whiteNoise;
            closeAudioModal();
        }

        function extractVideoID(url) {
            const regExp = /^.*(youtu.be\/|v\/|u\/\w\/|embed\/|watch\?v=|\&v=)([^#\&\?]*).*/;
            const match = url.match(regExp);
            return (match && match[2].length === 11) ? match[2] : null;
        }

        function drawHDBackground() {
            let skyTop = C.skyDayTop;
            let skyBottom = C.skyDayBottom;

            if (selectedTheme === 'aurora') {
                skyTop = C.skyAuroraTop;
                skyBottom = C.skyAuroraBottom;
            } else if (selectedTheme === 'space') {
                skyTop = C.skySpaceTop;
                skyBottom = C.skySpaceBottom;
            } else if (currentStage === 5) {
                skyTop = C.skySunsetTop;
                skyBottom = C.skySunsetBottom;
            } else if (currentStage >= 6) {
                skyTop = C.skySakuraTop;
                skyBottom = C.skySakuraBottom;
            }

            for (let y = 0; y < 52; y++) {
                let ratio = y / 52;
                ctx.fillStyle = interpolateColor(skyTop, skyBottom, ratio);
                ctx.fillRect(0, y * P, canvas.width, P);
            }

            clouds.forEach(c => {
                c.x += c.speed;
                if (c.x > GRID_SIZE) c.x = -c.w;

                let cy = Math.floor(c.y);

                for (let ox = 0; ox < c.w; ox++) {

                    for (let oy = 0; oy < h; oy++) {
                        drawPixel(cx + ox, cy - oy, C.cloudLight);
                        drawPixel(cx + ox, cy - oy + 1, C.cloudDark);
                    }
                }
            });

            for (let x = 0; x < GRID_SIZE; x++) {

                drawPixel(x, 53, (x % 3 === 0) ? C.grassMid : C.grassShadow);

                    let color = C.dirtMid;

                    if (pattern === 3) color = C.dirtDark;

                }
            }
        }

        function interpolateColor(color1, color2, factor) {
            let c1 = parseInt(color1.slice(1), 16);
            let c2 = parseInt(color2.slice(1), 16);

            let r1 = (c1 >> 16) & 0xff, g1 = (c1 >> 8) & 0xff, b1 = c1 & 0xff;
            let r2 = (c2 >> 16) & 0xff, g2 = (c2 >> 8) & 0xff, b2 = c2 & 0xff;

            let r = Math.round(r1 + factor * (r2 - r1));
            let g = Math.round(g1 + factor * (g2 - g1));
            let b = Math.round(b1 + factor * (b2 - b1));

            return `rgb(${r}, ${g}, ${b})`;
        }

        function drawOrganicTrunk(x, y, w, h) {
            let rim = C.woodRim, hi = C.woodHighlight, mid = C.woodMid, sh = C.woodShadow, dp = C.woodDeep;
            if (selectedTree === 'neon') { rim = C.neonRim; hi = C.neonHighlight; mid = C.neonMid; sh = C.neonShadow; dp = C.neonDeep; }
            if (selectedTree === 'golden') { rim = C.goldRim; hi = C.goldHighlight; mid = C.goldMid; sh = C.goldShadow; dp = C.goldDeep; }

            for (let i = 0; i < h; i++) {
                let cy = y - i;
                let curW = w + Math.floor(Math.sin(i * 0.2) * 1.5);
                if (i < 4) curW += (4 - i); 
                let cx = x - Math.floor(curW / 2);

                for (let j = 0; j < curW; j++) {
                    let color = mid;
                    if (j === 0) color = rim;
                    else if (j === 1) color = hi;
                    else if (j === curW - 2) color = sh;
                    else if (j === curW - 1) color = dp;
                    drawPixel(cx + j, cy, color);
                }
            }
        }

        function drawLeafCluster(cx, cy, r, leafType) {
            let rim = C.leafRim, hi = C.leafHighlight, mid = C.leafMid, sh = C.leafShadow, dp = C.leafDeep;

            if (leafType === 'autumn') { rim = C.autumnRim; hi = C.autumnHighlight; mid = C.autumnMid; sh = C.autumnShadow; dp = C.autumnDeep; }
            if (leafType === 'sakura') { rim = C.sakuraRim; hi = C.sakuraHighlight; mid = C.sakuraMid; sh = C.sakuraShadow; dp = C.sakuraDeep; }
            if (selectedTree === 'neon') { rim = C.neonRim; hi = C.neonHighlight; mid = C.neonMid; sh = C.neonShadow; dp = C.neonDeep; }
            if (selectedTree === 'golden') { rim = C.goldRim; hi = C.goldHighlight; mid = C.goldMid; sh = C.goldShadow; dp = C.goldDeep; }

            for (let dy = -r; dy <= r; dy++) {
                for (let dx = -r; dx <= r; dx++) {
                    let dist = Math.sqrt(dx * dx + dy * dy);
                    if (dist <= r + (Math.sin(dx * 3 + dy * 5) * 0.8)) {
                        let color = mid;
                        if (dy < -r * 0.4 && dx < 0) color = rim;
                        else if (dy < 0) color = hi;
                        else if (dy > r * 0.4 && dx > 0) color = dp;
                        else if (dy > 0) color = sh;

                        drawPixel(cx + dx, cy + dy, color);
                    }
                }
            }
        }

        function updateAmbientRain() {
            if (!isRaining) return;

            rainDurationCounter--;
            if (rainDurationCounter <= 0) {
                isRaining = false;
                nextRainInSeconds = getRandomRainInterval();
                ambientRainDrops = [];
                return;
            }

            ambientRainDrops.forEach(drop => {
                drop.y += drop.speed;
                if (drop.y > 52) {
                    drop.y = Math.random() * -10;
                    drop.x = Math.random() * GRID_SIZE;
                }
                drawPixel(drop.x, drop.y, C.rain);
                drawPixel(drop.x, drop.y - 1, C.rainTail);
            });
        }

        function spawnFloatingParticles() {
            if (currentStage >= 4 && Math.random() < 0.3) {
                let pColor = (currentStage >= 6) ? C.sakuraHighlight : C.leafHighlight;
                if (selectedTree === 'neon') pColor = C.neonRim;
                if (selectedTree === 'golden') pColor = C.goldHighlight;

                particles.push({
                    x: 32 + (Math.random() * 24 - 12),
                    y: 20 + (Math.random() * 20),
                    vx: (Math.random() - 0.5) * 0.3,
                    vy: Math.random() * 0.3 + 0.1,
                    color: pColor,
                    life: 100
                });
            }

            particles.forEach((p, index) => {
                p.x += p.vx;
                p.y += p.vy;
                p.life--;
                drawPixel(p.x, p.y, p.color);

                if (p.life <= 0 || p.y >= 52) {
                    particles.splice(index, 1);
                }
            });
        }

        function checkAndUpdateTreeStage() {
            let progressRatio = (TOTAL_SECONDS - secondsRemaining) / TOTAL_SECONDS;
            if (secondsRemaining === 0) progressRatio = 1.0;

            let newStage = Math.floor(progressRatio * 6);
            if (newStage > 6) newStage = 6;

            if (newStage !== currentStage) {
                currentStage = newStage;
                updateStatusText();
            }
        }

        function updateDisplay() {
            let m = Math.floor(secondsRemaining / 60);
            let s = secondsRemaining % 60;
            timerDisplay.innerText = `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
        }

        function processTimerTick() {
            const now = Date.now();
            secondsRemaining = Math.max(0, Math.ceil((targetEndTime - now) / 1000));

            if (isTimerRunning && nextRainInSeconds > 0) {
                nextRainInSeconds--;
                if (nextRainInSeconds <= 0 && !isRaining) {
                    triggerRainShower();
                }
            }

            updateDisplay();
            checkAndUpdateTreeStage();

            if (secondsRemaining <= 0) {
                clearInterval(timerInterval);
                isTimerRunning = false;
                startBtn.innerText = "▶ START";
                
                playBellChime();
                triggerConfetti();

                let earnedCoins = calculateRewards();
                statusText.innerText = `🎉 Session Complete! Earned +${earnedCoins} Eco Coins!`;

                if (isAudioLoaded && ytPlayer && ytPlayer.pauseVideo) ytPlayer.pauseVideo();
            }
        }

        function toggleTimer() {
            if (!isTimerRunning) {
                if (secondsRemaining <= 0) {
                    secondsRemaining = TOTAL_SECONDS;
                }

                targetEndTime = Date.now() + (secondsRemaining * 1000);
                timerInterval = setInterval(processTimerTick, 250);
                isTimerRunning = true;
                startBtn.innerText = "⏸ PAUSE";

                initAudioContext();
                if (audioCtx && audioCtx.state === 'suspended') {
                    audioCtx.resume();
                }

                if (isAudioLoaded && ytPlayer && ytPlayer.playVideo) ytPlayer.playVideo();
            } else {
                clearInterval(timerInterval);
                isTimerRunning = false;
                pausedDuringSession = true;
                startBtn.innerText = "▶ RESUME";

                if (isAudioLoaded && ytPlayer && ytPlayer.pauseVideo) ytPlayer.pauseVideo();
            }
        }

        function resetPomodoro() {
            clearInterval(timerInterval);
            isTimerRunning = false;
            pausedDuringSession = false;
            secondsRemaining = TOTAL_SECONDS;
            currentStage = 0;
            startBtn.innerText = "▶ START";

            updateDisplay();
            updateStatusText();

            if (isAudioLoaded && ytPlayer && ytPlayer.pauseVideo) ytPlayer.pauseVideo();
        }

            mainLoopId = requestAnimationFrame(mainLoop);
